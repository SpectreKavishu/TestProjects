    function loadCards() {

      const saved = localStorage.getItem(STORAGE_KEY);

      return saved ? JSON.parse(saved) : [...defaultCards];

    }

    

    function saveCards() {

      localStorage.setItem(STORAGE_KEY, JSON.stringify(cards));

    }

    

    let cards = loadCards();

    let currentType = 'image';

    let activeTag = null;

    let searchQuery = '';

    

    function renderCards() {

      const masonry = document.getElementById('masonry');

      let filtered = cards;

      

      if (activeTag) filtered = filtered.filter(c => c.tags.includes(activeTag));

      if (searchQuery) {

        const q = searchQuery.toLowerCase();

        filtered = filtered.filter(c => 

          c.title.toLowerCase().includes(q) ||

          c.desc.toLowerCase().includes(q) ||

          c.tags.some(t => t.toLowerCase().includes(q))

        );

      }

      

      if (filtered.length === 0) {

        masonry.innerHTML = `<div class="empty-state" style="column-span: all;"><h2>Nothing here yet</h2><p>Start adding things to your mind.</p></div>`;

        return;

      }

      

      masonry.innerHTML = filtered.map(card => {

        if (card.type === 'image') {

          return `<div class="card" onclick="openDetail(${card.id})">

            <div class="card-actions"><button class="card-action-btn" onclick="event.stopPropagation(); deleteCard(${card.id})">🗑</button></div>

            <img class="card-image" src="${card.url}" alt="${card.title}" loading="lazy">

            <div class="card-content"><div class="card-title">${card.title}</div><div class="card-desc">${card.desc}</div><div class="card-tags">${card.tags.map(t => `<span class="tag">${t}</span>`).join('')}</div></div>

          </div>`;

        } else if (card.type === 'note') {

          return `<div class="card card-note" onclick="openDetail(${card.id})">

            <div class="card-actions"><button class="card-action-btn" onclick="event.stopPropagation(); deleteCard(${card.id})">🗑</button></div>

            <div class="card-title">${card.title}</div><div class="card-desc">${card.desc.replace(/\n/g, '<br>')}</div><div class="card-tags">${card.tags.map(t => `<span class="tag">${t}</span>`).join('')}</div>

          </div>`;

        } else if (card.type === 'link') {

          return `<div class="card" onclick="openDetail(${card.id})">

            <div class="card-actions"><button class="card-action-btn" onclick="event.stopPropagation(); deleteCard(${card.id})">🗑</button></div>

            <div class="card-link"><div class="link-favicon">${card.favicon || '🔗'}</div><div class="link-info"><div class="card-title">${card.title}</div><div class="link-url">${card.url}</div></div></div>

          </div>`;

        }

    }

    

    function renderTags() {

      const allTags = [...new Set(cards.flatMap(c => c.tags))].sort();

        allTags.map(t => `<div class="filter-tag ${activeTag === t ? 'active' : ''}" onclick="setTag('${t}')">${t}</div>`).join('');

    }

    function openModal() { document.getElementById('modal').classList.add('active'); document.getElementById('inputTitle').focus(); }

    function closeModal() { document.getElementById('modal').classList.remove('active'); document.getElementById('inputTitle').value = ''; document.getElementById('inputUrl').value = ''; document.getElementById('inputDesc').value = ''; document.getElementById('inputTags').value = ''; }

    

    function selectType(type) {

      currentType = type;

      document.querySelectorAll('.type-option').forEach(el => el.classList.toggle('active', el.dataset.type === type));

      const urlGroup = document.getElementById('urlGroup');

      const urlLabel = urlGroup.querySelector('.form-label');

      const urlInput = document.getElementById('inputUrl');

      if (type === 'image') { urlGroup.style.display = 'block'; urlLabel.textContent = 'Image URL'; urlInput.placeholder = 'https://...'; }

      else if (type === 'link') { urlGroup.style.display = 'block'; urlLabel.textContent = 'URL'; urlInput.placeholder = 'https://example.com'; }

      else { urlGroup.style.display = 'none'; }

    }

    

    function addCard() {

      const title = document.getElementById('inputTitle').value.trim();

      const desc = document.getElementById('inputDesc').value.trim();

      const url = document.getElementById('inputUrl').value.trim();

      const tagsStr = document.getElementById('inputTags').value.trim();

      

      if (!title) { showToast('Please enter a title'); return; }

      

      const newCard = {

        id: Date.now(),

        type: currentType,

        title,

        desc: desc || '',

        tags: tagsStr ? tagsStr.split(',').map(t => t.trim().toLowerCase()).filter(Boolean) : [],

        date: new Date().toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })

      };

      

      if (currentType === 'image' || currentType === 'link') {

        newCard.url = url || (currentType === 'image' ? 'https://via.placeholder.com/600x400?text=No+Image' : '#');

      }

      if (currentType === 'link') {

        try { newCard.favicon = new URL(url).hostname[0].toUpperCase(); } catch { newCard.favicon = '🔗'; }

      }

      

      cards.unshift(newCard);

      saveCards();

      renderCards();

      renderTags();

      closeModal();

      showToast('Added to your mind');

    }

    

    function deleteCard(id) {

      if (confirm('Remove from your mind?')) {

        cards = cards.filter(c => c.id !== id);

        saveCards();

        renderCards();

        renderTags();

        showToast('Removed');

      }

    }

    

    function openDetail(id) {

      const card = cards.find(c => c.id === id);

      if (!card) return;

      const overlay = document.getElementById('detailOverlay');

      const detailCard = document.getElementById('detailCard');

      let html = '';

      if (card.type === 'image' && card.url) html += `<img class="detail-image" src="${card.url}" alt="${card.title}">`;

      html += `<div class="detail-content"><div class="detail-title">${card.title}</div><div class="detail-desc">${card.desc.replace(/\n/g, '<br>')}</div>`;

      if (card.type === 'link' && card.url) html += `<div style="margin-bottom: 16px;"><a href="${card.url}" target="_blank" style="color: #888; font-size: 14px; text-decoration: none; border-bottom: 1px solid #333; padding-bottom: 2px;">${card.url} →</a></div>`;

      html += `<div class="card-tags" style="margin-bottom: 20px;">${card.tags.map(t => `<span class="tag">${t}</span>`).join('')}</div>`;

      html += `<div class="detail-meta"><span>📅 ${card.date}</span><span>🏷️ ${card.type}</span></div></div>`;

      detailCard.innerHTML = html;

      overlay.classList.add('active');

      document.body.style.overflow = 'hidden';

    }

    

    function closeDetail(e) {

      if (e && e.target !== e.currentTarget) return;

      document.getElementById('detailOverlay').classList.remove('active');

      document.body.style.overflow = '';

    }

    

    function showToast(msg) {

      const toast = document.getElementById('toast');

      toast.textContent = msg;

      toast.classList.add('show');

      setTimeout(() => toast.classList.remove('show'), 2500);

    }

    

    document.addEventListener('keydown', (e) => {

      if (e.key === 'Escape') { closeModal(); closeDetail(); }

      if ((e.metaKey || e.ctrlKey) && e.key === 'k') { e.preventDefault(); document.getElementById('searchBox').focus(); }

      if ((e.metaKey || e.ctrlKey) && e.key === 'n') { e.preventDefault(); openModal(); }

    });

    

    renderTags();

    renderCards();
